cmake_minimum_required(VERSION 3.16)
project(Pi0Reconstruction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED ENV{OFFLINE_MAIN})
  message(FATAL_ERROR "Source a fixed sPHENIX release first")
endif()
if(DEFINED PHOTON_CONFIGURED_PREFIX AND NOT PHOTON_CONFIGURED_PREFIX STREQUAL "$ENV{OFFLINE_MAIN}")
  message(FATAL_ERROR "This build directory belongs to another sPHENIX release")
endif()
set(PHOTON_CONFIGURED_PREFIX "$ENV{OFFLINE_MAIN}" CACHE INTERNAL "Configured sPHENIX release")
find_package(ROOT REQUIRED COMPONENTS Core Hist Tree)
add_library(Pi0Reconstruction SHARED Pi0Reconstruction.cc ShowerShapeCalculator.cc TowerClusterEnergyAudit.cc)
set_target_properties(Pi0Reconstruction PROPERTIES VERSION 0.0.0 SOVERSION 0)
target_include_directories(Pi0Reconstruction PRIVATE "$ENV{OFFLINE_MAIN}/include")
target_link_libraries(Pi0Reconstruction PRIVATE ROOT::Core ROOT::Hist ROOT::Tree)
foreach(name phool SubsysReco ffaobjects ffarawobjects calo_io globalvertex track_io intt)
  find_library(${name}_LIBRARY NAMES ${name} HINTS "$ENV{OFFLINE_MAIN}/lib" "$ENV{OFFLINE_MAIN}/lib64" REQUIRED)
  target_link_libraries(Pi0Reconstruction PRIVATE "${${name}_LIBRARY}")
endforeach()
install(TARGETS Pi0Reconstruction LIBRARY DESTINATION lib)
install(FILES Pi0Reconstruction.h ShowerShapeCalculator.h TowerClusterEnergyAudit.h DESTINATION include/pi0reconstruction)
